import java.util.Scanner;

// Classe Swapper com atributos privados e método de troca
class Swapper {
    private float x;
    private float y;

    // Construtor padrão
    public Swapper() {
    }

    // Getters e Setters para x
    public float getX() {
        return x;
    }

    public void setX(float x) {
        this.x = x;
    }

    // Getters e Setters para y
    public float getY() {
        return y;
    }

    public void setY(float y) {
        this.y = y;
    }

    // Método void swap() que troca os valores de x e y
    public void swap() {
        float temp = this.x;
        this.x = this.y;
        this.y = temp;
    }
}

// Classe SwapperDemo contendo o método main
public class SwapperDemo {
    public static void main(String[] args) {
        Scanner leitor = new Scanner(System.in);

        // Instanciação do objeto troca do tipo Swapper
        Swapper troca = new Swapper();

        // Leitura dos valores de x e y pelo teclado
        System.out.println("--- EXERCÍCIO 2: CLASSE SWAPPER ---");
        System.out.print("Digite o valor para x (float): ");
        float valorX = leitor.nextFloat();

        System.out.print("Digite o valor para y (float): ");
        float valorY = leitor.nextFloat();

        // Inicializando os atributos usando os métodos set
        troca.setX(valorX);
        troca.setY(valorY);

        System.out.println("\n--- VALORES INICIAIS ---");
        System.out.println("x: " + troca.getX());
        System.out.println("y: " + troca.getY());

        // Invocando o método swap() para realizar a troca
        troca.swap();

        System.out.println("\n--- VALORES APÓS SWAP() ---");
        System.out.println("x: " + troca.getX());
        System.out.println("y: " + troca.getY());

        leitor.close();
    }
}

Exercício – Classe Retangulo
(Deitel 8.14) Crie a classe Retângulo mais sofisticada. A
classe deve armazenar as coordenadas cartesianas dos
quatro vértices do retângulo.
O construtor chama o método set que aceita quatro
conjuntos de coordenadas (8 valores no total: 4 xs e 4 ys) e
verifica se cada um deles está no primeiro quadrante,
limitados a um valor superior igual a 20.0.
O método set também verifica se as coordenadas
fornecidas especificam um retângulo. A classe deve ter
também métodos para calcular a comprimento, a largura e
o perímetro do retângulo. O comprimento é a maior das
duas dimensões. 

1. **Atributos**: A classe armazena os **4 vértices** do retângulo via 8 coordenadas de ponto flutuante (`x1, y1, x2, y2, x3, y3, x4, y4`).
2. **Construtor e Validação (`set`)**:
   - O construtor invoca o método `set(...)`, que aceita os 8 valores.
   - **Validação de Limites**: Garante que cada coordenada pertença ao **primeiro quadrante** (\\(x \ge 0.0, y \ge 0.0\\)) e esteja limitada ao valor máximo de **\\(20.0\\)**.
   - **Validação Geométrica**: Verifica matematicamente se as 4 coordenadas formam de fato um retângulo (testando se as duas diagonais possuem comprimentos iguais e se cruzam no mesmo ponto médio). Caso as coordenadas sejam inválidas, o programa exibe uma mensagem explicativa e define um retângulo padrão \\(1 \times 1\\).
3. **Cálculos das Dimensões**:
   - **`obterComprimento()`**: Identifica e retorna a **maior das duas dimensões**.
   - **`obterLargura()`**: Identifica e retorna a **menor das duas dimensões**.
   - **`obterPerimetro()`**: Retorna \\(2 \times (\text{comprimento} + \text{largura})\\).

---

### **Código-Fonte Completo (`RetanguloExercicio.java`)**

// Classe Retangulo desenvolvida conforme o exercício (Deitel 8.14)
class Retangulo {
    // Coordenadas dos quatro vértices
    private double x1, y1;
    private double x2, y2;
    private double x3, y3;
    private double x4, y4;

    // Construtor padrão (inicia com um retângulo padrão 1x1)
    public Retangulo() {
        set(0.0, 0.0, 1.0, 0.0, 1.0, 1.0, 0.0, 1.0);
    }

    // Construtor parametrizado (chama o método set)
    public Retangulo(double x1, double y1, double x2, double y2, double x3, double y3, double x4, double y4) {
        set(x1, y1, x2, y2, x3, y3, x4, y4);
    }

    // Método set mestre para configurar e validar todas as coordenadas
    public void set(double x1, double y1, double x2, double y2, double x3, double y3, double x4, double y4) {
        // 1. Verifica se todas as coordenadas estão no primeiro quadrante (>= 0.0) e limitadas a 20.0
        boolean coordsValidas = validaLimite(x1) && validaLimite(y1) &&
                               validaLimite(x2) && validaLimite(y2) &&
                               validaLimite(x3) && validaLimite(y3) &&
                               validaLimite(x4) && validaLimite(y4);

        if (!coordsValidas) {
            System.out.println("Erro: Coordenadas devem estar no primeiro quadrante (0.0 a 20.0). Retângulo padrão (1x1) definido.");
            configuraPadrao();
            return;
        }

        // 2. Verifica se as coordenadas formam um retângulo
        if (!eRetangulo(x1, y1, x2, y2, x3, y3, x4, y4)) {
            System.out.println("Erro: As coordenadas fornecidas não especificam um retângulo válido. Retângulo padrão (1x1) definido.");
            configuraPadrao();
            return;
        }

        // Se passar nas validações, armazena os valores
        this.x1 = x1; this.y1 = y1;
        this.x2 = x2; this.y2 = y2;
        this.x3 = x3; this.y3 = y3;
        this.x4 = x4; this.y4 = y4;
    }

    private boolean validaLimite(double val) {
        return val >= 0.0 && val <= 20.0;
    }

    private void configuraPadrao() {
        this.x1 = 0.0; this.y1 = 0.0;
        this.x2 = 1.0; this.y2 = 0.0;
        this.x3 = 1.0; this.y3 = 1.0;
        this.x4 = 0.0; this.y4 = 1.0;
    }

    private boolean eRetangulo(double x1, double y1, double x2, double y2, double x3, double y3, double x4, double y4) {
        return verificaDiagonais(x1, y1, x2, y2, x3, y3, x4, y4) ||
               verificaDiagonais(x1, y1, x3, y3, x2, y2, x4, y4) ||
               verificaDiagonais(x1, y1, x4, y4, x2, y2, x3, y3);
    }

    private boolean verificaDiagonais(double xa, double ya, double xb, double yb, double xc, double yc, double xd, double yd) {
        double midX_AB = (xa + xb) / 2.0;
        double midY_AB = (ya + yb) / 2.0;
        double midX_CD = (xc + xd) / 2.0;
        double midY_CD = (yc + yd) / 2.0;
        
        double epsilon = 0.0001;
        boolean mesmosPontosMedios = Math.abs(midX_AB - midX_CD) < epsilon && Math.abs(midY_AB - midY_CD) < epsilon;
        
        double distSq_AB = Math.pow(xa - xb, 2) + Math.pow(ya - yb, 2);
        double distSq_CD = Math.pow(xc - xd, 2) + Math.pow(yc - yd, 2);
        
        boolean mesmosComprimentos = Math.abs(distSq_AB - distSq_CD) < epsilon;
        boolean diagonalValida = distSq_AB > epsilon;
        
        return mesmosPontosMedios && mesmosComprimentos && diagonalValida;
    }

    private double[] obterLados() {
        double d12 = Math.sqrt(Math.pow(x1 - x2, 2) + Math.pow(y1 - y2, 2));
        double d13 = Math.sqrt(Math.pow(x1 - x3, 2) + Math.pow(y1 - y3, 2));
        double d14 = Math.sqrt(Math.pow(x1 - x4, 2) + Math.pow(y1 - y4, 2));

        double lado1 = d12;
        double lado2 = d13;
        double diagonal = d14;

        if (lado1 > diagonal) { double temp = lado1; lado1 = diagonal; diagonal = temp; }
        if (lado2 > diagonal) { double temp = lado2; lado2 = diagonal; diagonal = temp; }
        
        return new double[]{lado1, lado2};
    }

    public double obterLargura() {
        double[] lados = obterLados();
        return Math.min(lados, lados);
    }

    public double obterComprimento() {
        double[] lados = obterLados();
        return Math.max(lados, lados);
    }

    public double obterPerimetro() {
        return 2 * (obterLargura() + obterComprimento());
    }

    public double getX1() { return x1; }
    public double getY1() { return y1; }
    public double getX2() { return x2; }
    public double getY2() { return y2; }
    public double getX3() { return x3; }
    public double getY3() { return y3; }
    public double getX4() { return x4; }
    public double getY4() { return y4; }
}

public class RetanguloExercicio {
    public static void main(String[] args) {
        System.out.println("--- Retângulo Válido (4x3) ---");
        Retangulo r1 = new Retangulo(0.0, 0.0, 4.0, 0.0, 4.0, 3.0, 0.0, 3.0);
        exibirInformacoes(r1);

        System.out.println("\n--- Tentando Coordenada Negativa (-1,2) ---");
        Retangulo r2 = new Retangulo(0.0, 0.0, 2.0, 1.0, 1.0, 3.0, -1.0, 2.0);
        exibirInformacoes(r2);

        System.out.println("\n--- Corrigindo Coordenada para 1º Quadrante ---");
        r2.set(1.0, 0.0, 3.0, 1.0, 2.0, 3.0, 0.0, 2.0);
        exibirInformacoes(r2);
    }

    public static void exibirInformacoes(Retangulo r) {
        System.out.printf("Vértices: P1(%.1f,%.1f), P2(%.1f,%.1f), P3(%.1f,%.1f), P4(%.1f,%.1f)\n",
                r.getX1(), r.getY1(), r.getX2(), r.getY2(), r.getX3(), r.getY3(), r.getX4(), r.getY4());
        System.out.printf("Largura      : %.4f\n", r.obterLargura());
        System.out.printf("Comprimento  : %.4f\n", r.obterComprimento());
        System.out.printf("Perímetro    : %.4f\n", r.obterPerimetro());
    }

package a04e04;

/**
 *
 * @author unifmsegura
 */
public class ContaCorrente {
    private double saldo;
    
    // construtor que inicia a conta com saldo zero
    public ContaCorrente(){
        this.saldo = 0.0;
    }
    
    public ContaCorrente (double saldoInicial){
        if (saldoInicial >= 0.0){
            this.saldo = saldoInicial;
        } else {
            this.saldo = 0.0;
        }
    }
    
    // Alt + Insert, depois selecionar Getter and Setter
    // Botão direito do mouse, Insert code

    public double getSaldo() {
        return saldo;
    }

    public void setSaldo(double saldo) {
        this.saldo = saldo;
    }
    
    // depositar
    public void depositar (double valor){
        if (valor > 0.0){
            this.saldo += valor;
            System.out.printf("Deposito de R$ %.2f realizado. Novo Saldo: R$ %.2f\n", valor, this.saldo);
        } else {
            System.out.println("O valor de depositivo deve ser positivo.");
        }
    }
    
    // sacar
    public boolean sacar (double valor) {
        if (valor >= 0.0){
            this.saldo -= valor;
            System.out.printf("Saque de R$ %.2f realizado. Novo Saldo: R$ %.2f\n", valor, this.saldo);
            return true;
        } else {
            System.out.printf("Saldo insuficiente para sacar R$ %.2f. Saldo disponivel: R$ %.2f\n", valor, this.saldo);
            return false;
        }
    }
    
    @Override
    public String toString(){
        return "Saldo=" + this.saldo;
    }
}

/**
 * Aplicativo de gerenciamento de Agenda utilizando ArrayList e loop interativo.
 */
public class AgendaExercicio {
    public static void main(String[] args) {
        Scanner leitor = new Scanner(System.in);
        ArrayList<Pessoa> agenda = new ArrayList<>();

        System.out.println("=============================================");
        System.out.println("         SISTEMA DE AGENDA TELEFÔNICA        ");
        System.out.println("=============================================");

        // Loop infinito até que a entrada seja 'q'
        while (true) {
            System.out.println("\n--- Escolha uma opção: ---");
            System.out.println(" n : Cadastrar nova pessoa");
            System.out.println(" d : Deletar pessoa por nome");
            System.out.println(" p : Imprimir toda a agenda");
            System.out.println(" q : Sair do programa");
            System.out.print("Sua opção: ");
            
            // Lê a opção removendo espaços extras e convertendo para minúsculo
            String opcao = leitor.nextLine().trim().toLowerCase();

            if (opcao.equals("q")) {
                System.out.println("\nEncerrando o programa da Agenda. Obrigado!");
                break; // Quebra o loop infinito e encerra o main
            }

            switch (opcao) {
                case "n":
                    System.out.println("\n[NOVO CADASTRO]");
                    System.out.print("Entre com o nome da pessoa: ");
                    String nome = leitor.nextLine().trim();
                    
                    System.out.print("Entre com o telefone da pessoa: ");
                    String telefone = leitor.nextLine().trim();

                    // Cria e insere o objeto dinamicamente no ArrayList
                    Pessoa novaPessoa = new Pessoa(nome, telefone);
                    agenda.add(novaPessoa);
                    System.out.println("Pessoa cadastrada com sucesso! " + novaPessoa);
                    break;

                case "d":
                    System.out.println("\n[REMOVER REGISTRO]");
                    if (agenda.isEmpty()) {
                        System.out.println("A agenda está vazia! Não há ninguém para excluir.");
                        break;
                    }

                    System.out.print("Entre com o nome da pessoa que será excluída: ");
                    String nomeParaDeletar = leitor.nextLine().trim();

                    boolean removido = false;
                    // Percorre o ArrayList buscando a pessoa pelo nome
                    for (int i = 0; i < agenda.size(); i++) {
                        if (agenda.get(i).getNome().equalsIgnoreCase(nomeParaDeletar)) {
                            Pessoa pessoaRemovida = agenda.remove(i); // Remove e recupera o objeto
                            System.out.println("Registro removido com sucesso: " + pessoaRemovida);
                            removido = true;
                            break; // Interrompe a busca, pois assumimos nomes únicos
                        }
                    }

                    if (!removido) {
                        System.out.println("Erro: Nenhuma pessoa encontrada com o nome '" + nomeParaDeletar + "'.");
                    }
                    break;

                case "p":
                    System.out.println("\n[IMPRIMIR AGENDA]");
                    if (agenda.isEmpty()) {
                        System.out.println("=============================================");
                        System.out.println("          A AGENDA ESTÁ VAZIA!               ");
                        System.out.println("=============================================");
                    } else {
                        System.out.println("=============================================");
                        System.out.println("               CONTATOS SALVOS               ");
                        System.out.println("=============================================");
                        // Percorre a coleção utilizando o loop for-each moderno
                        for (Pessoa p : agenda) {
                            System.out.println(p);
                        }
                        System.out.println("=============================================");
                        System.out.println("Total de registros: " + agenda.size());
                    }
                    break;

                default:
                    System.out.println("Opção inválida! Digite apenas n, d, p ou q.");
                    break;
            }
        }

        leitor.close();
    }
}

class Carro {
    // Atributos de Identificacao
    private String marca;
    private String modelo;

    // Atributos de Composicao de Parte Unica
    private Motor motor;
    private Volante volante;
    private Transmissao transmissao;
    private Chassi chassi;
    private ArCondicionado arCondicionado;
    private Painel painel;

    // Atributos de Composicao de Multiplas Partes (Gerenciados por ArrayList)
    private ArrayList<Roda> rodas;
    private ArrayList<Porta> portas;
    private ArrayList<Banco> bancos;
    private ArrayList<Farol> farois;

    /**
     * Construtor parametrizado para Carro.
     * Os objetos internos de multipla quantidade (como Rodas, Portas, Bancos, Farois)
     * sao criados e controlados diretamente pela classe Carro, caracterizando COMPOSICAO.
     */
    public Carro(String marca, String modelo, Motor motor, Volante volante, Transmissao transmissao,
                 Chassi chassi, ArCondicionado arCondicionado, Painel painel,
                 int aroRodas, String marcaRodas, int qtdPortas, String materialBancos, String corBancos) {
        
        this.marca = marca;
        this.modelo = modelo;
        this.motor = motor;
        this.volante = volante;
        this.transmissao = transmissao;
        this.chassi = chassi;
        this.arCondicionado = arCondicionado;
        this.painel = painel;

        // Instanciacao dos ArrayLists (O ciclo de vida pertence ao Carro)
        this.rodas = new ArrayList<>();
        for (int i = 0; i < 4; i++) {
            this.rodas.add(new Roda(aroRodas, marcaRodas));
        }

        this.portas = new ArrayList<>();
        this.portas.add(new Porta("Frente Esquerda", true));
        this.portas.add(new Porta("Frente Direita", true));
        if (qtdPortas == 4) {
            this.portas.add(new Porta("Traseira Esquerda", true));
            this.portas.add(new Porta("Traseira Direita", true));
        }

        this.bancos = new ArrayList<>();
        int totalBancos = (qtdPortas == 4) ? 5 : 4;
        for (int i = 0; i < totalBancos; i++) {
            this.bancos.add(new Banco(materialBancos, corBancos));
        }

        this.farois = new ArrayList<>();
        this.farois.add(new Farol("LED", true));
        this.farois.add(new Farol("LED", true));
    }

    public void setMarca(String marca) { this.marca = marca; }

    public String getModelo() { return modelo; }
    public void setModelo(String modelo) { this.modelo = modelo; }

    public void setMotor(Motor motor) { this.motor = motor; }

    public void setFarois(ArrayList<Farol> farois) { this.farois = farois; }

    // Representacao em String formatada do Carro e de todas as suas partes componentes
    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder();
        sb.append(String.format("=========================================\n"));
        sb.append(String.format("CARRO COMPLETO: %s %s\n", marca, modelo));
        sb.append(String.format("=========================================\n"));
        sb.append("Partes Singulares:\n");
        sb.append("  - ").append(motor).append("\n");
        sb.append("  - ").append(volante).append("\n");
        sb.append("  - ").append(transmissao).append("\n");
        sb.append("  - ").append(chassi).append("\n");
        sb.append("  - ").append(arCondicionado).append("\n");
        sb.append("  - ").append(painel).append("\n");
        
        sb.append("Colecoes de Partes (ArrayLists):\n");
        sb.append("  * Rodas (").append(rodas.size()).append(" itens):\n");
        for (Roda r : rodas) sb.append("     - ").append(r).append("\n");
        
        sb.append("  * Portas (").append(portas.size()).append(" itens):\n");
        for (Porta p : portas) sb.append("     - ").append(p).append("\n");
        
        sb.append("  * Bancos (").append(bancos.size()).append(" itens):\n");
        for (Banco b : bancos) sb.append("     - ").append(b).append("\n");
        
        sb.append("  * Farois (").append(farois.size()).append(" itens):\n");
        for (Farol f : farois) sb.append("     - ").append(f).append("\n");
        
        return sb.toString();
    }

    /**
     * @param args the command line arguments
     */
    public static void main(String[] args) {
        Scanner leitor = new Scanner(System.in);
        ArrayList<Aluno> listaAlunos = new ArrayList<>();

        System.out.println("=================================================");
        System.out.println("       EXERCÍCIO 1 - HIERARQUIA DE ALUNO        ");
        System.out.println("=================================================\n");

        // Entrada de dados via teclado
        System.out.println("Entre com os dados do aluno para o construtor parametrizado:");
        
        System.out.print("Digite o nome: ");
        String nome = leitor.nextLine().trim();

        System.out.print("Digite o sobrenome: ");
        String sobrenome = leitor.nextLine().trim();

        System.out.print("Digite a idade: ");
        int idade = leitor.nextInt();
        leitor.nextLine(); // Limpa o buffer do teclado

        System.out.print("Digite o curso: ");
        String curso = leitor.nextLine().trim();
        System.out.println();

        // 1. Instanciação de objeto sem passar argumentos (construtor padrão)
        Aluno aluno1 = new Aluno();

        // 2. Instanciação de objeto com o construtor parametrizado
        Aluno aluno2 = new Aluno(nome, sobrenome, idade, curso);

        // Adiciona ambos no ArrayList
        listaAlunos.add(aluno1);
        listaAlunos.add(aluno2);

        // Exibição do conteúdo do ArrayList utilizando o método print()
        System.out.println("=================================================");
        System.out.println("          CONTEÚDO DO ARRAYLIST DE ALUNOS        ");
        System.out.println("=================================================");
        
        for (int i = 0; i < listaAlunos.size(); i++) {
            System.out.println("Aluno " + (i + 1) + ":");
            listaAlunos.get(i).print();
        }

        leitor.close();
    }
}

class Aluno extends Pessoa {
    private String curso;

    /**
     * Construtor padrão (sem argumentos) que invoca o construtor da superclasse.
     */
    public Aluno() {
        super();
        this.curso = "Não informado";
    }

    /**
     * Construtor parametrizado que utiliza super() para reaproveitar
     * a inicialização dos atributos de Pessoa.
     * 
     * @param nome Nome do aluno
     * @param sobrenome Sobrenome do aluno
     * @param idade Idade do aluno
     * @param curso Curso em que o aluno está matriculado
     */
    public Aluno(String nome, String sobrenome, int idade, String curso) {
        super(nome, sobrenome, idade);
        this.curso = curso;
    }

    public String getCurso() {
        return curso;
    }

    public void setCurso(String curso) {
        this.curso = curso;
    }

    /**
     * Método print() conforme especificado no diagrama de classes da aula.
     */
    public void print() {
        System.out.printf("Nome Completo : %s %s\n", getNome(), getSobreNome());
        System.out.printf("Idade         : %d anos\n", getIdade());
        System.out.printf("Curso         : %s\n", getCurso());
        System.out.println("-------------------------------------------------");
    }

    @Override
    public String toString() {
        return String.format("Aluno: %s %s | Idade: %d | Curso: %s", 
                getNome(), getSobreNome(), getIdade(), getCurso());
    }
}

            this.balance = initialBalance;
        } else {
            this.balance = 0.0;
            System.out.println("Erro: O saldo inicial era inválido. Saldo configurado para R$ 0.00.");
        }
    }

    public void credit(double amount) {
        if (amount > 0.0) {
            balance += amount;
        }
    }

    public boolean debit(double amount) {
        if (amount <= balance) {
            balance -= amount;
            return true;
        } else {
            System.out.println("Saldo insuficiente.");
            return false;
        }
    }

    public double getBalance() {
        return balance;
    }
}

    public CheckingAccount(double initialBalance, double fee) {
        super(initialBalance);
        this.fee = fee;
    }

    @Override
    public void credit(double amount) {
        super.credit(amount);
        balance -= fee; // Cobra a taxa de transação após o crédito
    }

    @Override
    public boolean debit(double amount) {
        boolean success = super.debit(amount);
        if (success) {
            balance -= fee; // Cobra a taxa de transação apenas se o saque for realizado
        }
        return success;
    }
}

 *
 * @author unifmsegura
 */
class Trapezio extends Quadrilatero {

    public Trapezio(Ponto p1, Ponto p2, Ponto p3, Ponto p4) {
        super(p1, p2, p3, p4);
    }

    public double getBase1() {
        return Math.hypot(getP2().getX() - getP1().getX(), getP2().getY() - getP1().getY());
    }

    public double getBase2() {
        return Math.hypot(getP3().getX() - getP4().getX(), getP3().getY() - getP4().getY());
    }

    public double getAltura() {
        if (getP1().getY() == getP2().getY()) {
            return Math.abs(getP1().getY() - getP4().getY());
        } else {
            return Math.abs(getP1().getX() - getP4().getX());
        }
    }

    // O cálculo de área do trapézio serve também para todas as subclasses
    public double calcularArea() {
        return ((getBase1() + getBase2()) * getAltura()) / 2.0;
    }
}

// Classe Abstrata base para a hierarquia de formas geométricas
abstract class Forma {

    private String cor;

        this.nome = nome;
        this.cor = cor;
    }

        return cor;

        this.cor = cor;

    public abstract double calcularArea();
    public abstract double calcularPerimetro();

    @Override
    public String toString() {
        return String.format("%s (Cor: %s) | Área: %.2f | Perímetro: %.2f", 
                nome, cor, calcularArea(), calcularPerimetro());

## 📌 QUESTÃO 1: Associação de Blocos e Descrições (Conceitos Fundamentais)

| Termo | Descrição Correta | Exemplo em Java |
| :--- | :--- | :--- |
| **Classe** | Modelo, molde ou *template* a partir do qual os objetos são criados. Define quais atributos e métodos os objetos terão. | `public class Carro { ... }` |
| **Atributo** | Variável de instância pertencente a uma classe que representa as características, propriedades ou dados do objeto. | `private String cor;` |
| **Método** | Função-membro que define os comportamentos, ações ou operações que um objeto pode realizar. | `public void acelerar() { ... }` |
| **Hierarquia** | Estrutura de organização baseada em **Herança**, relacionando superclasses (genéricas) e subclasses (específicas). | `Carro extends Veiculo` |
| **Polimorfismo** | Capacidade de tratar objetos de diferentes subclasses como pertencentes à mesma superclasse, onde cada um responde ao mesmo método de forma adequada ao seu tipo. | `Animal a = new Pato();` <br>`a.emitirSom(); // Quack` |

        return quantidadePortas;
    }

## 📌 QUESTÕES 4, 5 e 6: Polimorfismo e a Hierarquia `Pato` / `Animal`

### Conceito do Som do Pato ("Quack" vs "Miau")
A superclasse `Animal` define o método genérico `emitirSom()`. Cada subclasse sobrescreve o método com a anotação **`@Override`**:

```java
// Superclasse
class Animal {
    public void emitirSom() {
        System.out.println("Som genérico de animal");
    }
}

// Subclasse Gato
class Gato extends Animal {
    @Override
    public void emitirSom() {
        System.out.println("Miau");
    }
}
```

## 💡 DICAS RÁPIDAS PARA A PROVA
1. **Atributos são sempre `private`** (ou `protected` quando houver herança).
2. **Nomes de Construtores** são **exatamente iguais** ao nome da classe e **não possuem tipo de retorno** (nem `void`).
3. **Uso do `this`**: Refere-se ao atributo da própria instância (ex: `this.nome = nome`).
4. **Instanciação**: Utiliza a palavra-chave `new` (ex: `Pato p = new Pato();`).
5. **Herança**: Usa `extends` na classe e `super(...)` no construtor da subclasse.
6. **Polimorfismo**: Usa a anotação `@Override` antes do método redefinido na subclasse.

    }

Questões 4, 5 e 6: Polimorfismo e a Hierarquia Pato / Animal
Conceito de Polimorfismo e Sobrescrita (@Override)
No polimorfismo, a superclasse Animal possui um método genérico (ex: emitirSom()), e cada subclasse reescreve o método com a anotação @Override 8.O método polimórfico garante que um Pato execute "quack" e um Gato execute "miau" 7, 8:
class Animal {
    public void emitirSom() {
        System.out.println("Som genérico de animal");
    }
}


class Gato extends Animal {
    @Override
    public void emitirSom() {
        System.out.println("Miau");
    }
}


class Pato extends Animal {
    // Questão 5: Adicionar 2 variáveis
    private String corPena;
    private double tamanhoBico;


    // Questão 6: Adicionar mais 2 variáveis (totalizando 4)
    private boolean voa;
    private int quantidadeOvos;


    public Pato(String corPena, double tamanhoBico, boolean voa, int quantidadeOvos) {
        this.corPena = corPena;
        this.tamanhoBico = tamanhoBico;
        this.voa = voa;
        this.quantidadeOvos = quantidadeOvos;
    }


    @Override
    public void emitirSom() {
        System.out.println("Quack"); // Sobrescrita correta do som do Pato
    }
}
Execução no main (Upcasting)
Ao declarar como Animal e instanciar como Pato, o Java chama a versão do método da classe real instanciada 7, 8:
Animal a = new Pato("Branca", 3.5, true, 4);
a.emitirSom(); // Imprime "Quack" por polimorfismo

Questões 7, 8 e 9: Escrita de Código
Nas questões 8 e 9, por serem sem herança nem polimorfismo, o foco é a estrutura clássica de Encapsulamento (atributos privados, construtores e métodos get/set) 1, 9, 14:
public class Produto {
    // 1. Atributos privados
    private String nome;
    private double preco;


    // 2. Construtor Padrão
    public Produto() {}


    // 3. Construtor Parametrizado
    public Produto(String nome, double preco) {
        this.nome = nome;
        this.preco = preco;
    }


    // 4. Métodos Get e Set
    public String getNome() { return nome; }
    public void setNome(String nome) { this.nome = nome; }


    public double getPreco() { return preco; }
    public void setPreco(double preco) { this.preco = preco; }


    // 5. Instanciação na classe principal (main)
    public static void main(String[] args) {
        Produto p1 = new Produto("Caderno", 25.00);
        System.out.println("Produto: " + p1.getNome() + " | Preço: R$ " + p1.getPreco());
    }
}
💡 Dica final: Na hora de escrever códigos simples no NetBeans, lembre-se de declarar atributos como private, fornecer os métodos get/set e usar this dentro dos construtores e setters 1, 9.
